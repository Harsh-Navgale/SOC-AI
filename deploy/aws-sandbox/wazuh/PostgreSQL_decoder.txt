<!--
  PostgreSQL decoders
  Expected postgresql.conf:
    log_line_prefix = '%m [%p] user=%u,db=%d,client=%r,app=%a '
    log_statement   = 'all'    (or 'mod' for DDL + DML only)
-->

<decoder name="postgresql">
  <prematch type="pcre2">\[\d+\]\s+user=[^,]*,db=</prematch>
</decoder>

<!-- Connection context + severity (LOG / STATEMENT / ERROR / FATAL ...) -->
<decoder name="postgresql_child">
  <parent>postgresql</parent>
  <regex type="pcre2">user=([^,]*),db=([^,]*),client=([^,]*),app=(.*?)\s+([A-Z]+):</regex>
  <order>db_user,database,client,app,pg_level</order>
</decoder>

<!-- INSERT -->
<decoder name="postgresql_child">
  <parent>postgresql</parent>
  <regex type="pcre2">(?i)(?:LOG:\s+(?:statement|execute\s+[^:]+)|STATEMENT):\s+(insert)\s+into\s+((?:"[^"]+"|[\w$]+)(?:\.(?:"[^"]+"|[\w$]+))?)</regex>
  <order>db_action,db_table</order>
</decoder>

<!-- UPDATE / DELETE / TRUNCATE -->
<decoder name="postgresql_child">
  <parent>postgresql</parent>
  <regex type="pcre2">(?i)(?:LOG:\s+(?:statement|execute\s+[^:]+)|STATEMENT):\s+(update|delete\s+from|truncate(?:\s+table)?)\s+(?:only\s+)?((?:"[^"]+"|[\w$]+)(?:\.(?:"[^"]+"|[\w$]+))?)</regex>
  <order>db_action,db_table</order>
</decoder>

<!-- CREATE / DROP / ALTER TABLE -->
<decoder name="postgresql_child">
  <parent>postgresql</parent>
  <regex type="pcre2">(?i)(?:LOG:\s+(?:statement|execute\s+[^:]+)|STATEMENT):\s+(create|drop|alter)\s+(?:(?:temp|temporary|unlogged)\s+)?table\s+(?:if\s+(?:not\s+)?exists\s+)?(?:only\s+)?((?:"[^"]+"|[\w$]+)(?:\.(?:"[^"]+"|[\w$]+))?)</regex>
  <order>db_action,db_table</order>
</decoder>

<!-- CREATE / DROP / ALTER DATABASE -->
<decoder name="postgresql_child">
  <parent>postgresql</parent>
  <regex type="pcre2">(?i)(?:LOG:\s+(?:statement|execute\s+[^:]+)|STATEMENT):\s+(create|drop|alter)\s+database\s+(?:if\s+(?:not\s+)?exists\s+)?("[^"]+"|[\w$]+)</regex>
  <order>db_action,target_db</order>
</decoder>
